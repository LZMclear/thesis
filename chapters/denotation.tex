\begin{denotation}

\item [${x}$] 标量${x}$
\item [${\bf{x}}$] 列矢量${\bf{x}}$
\item [${\bf{X}}$] 矩阵${\bf{X}}$
\item [${\bf{X}}_{m,n}$] 矩阵${\bf{X}}$的第$m$行第$n$列元素
\item [${\bf{I}}_N$] 维度为$N \times N$的单位矩阵
\item [$|\cdot|$] 矢量/矩阵的行列式或标量的模
\item [$\|\cdot\|$] 二范数操作
\item [$H(\cdot)$] 差分熵操作
\item [$\mathrm{tr}(\cdot)$] 迹操作
\item [$\mathbb{E}\{\cdot\}$] 取期望操作
\item [$\mathrm{Var}\{\cdot\}$] 取方差操作
\item [$(\cdot)^*$] 矢量/矩阵的共轭
\item [$(\cdot)^T$] 矢量/矩阵的转置
\item [$(\cdot)^{-1}$] 矢量/矩阵的逆阵
\item [$(\cdot)^H$] 矢量/矩阵的共轭转置
\item [$\mathbb{C}^{M \times N}$] 复数域上的$M \times N$维矩阵
\item [$\mathcal{CN}(\mathbf{0}, \Psi)$] 均值为$\mathbf{0}$、协方差矩阵为$\Psi$的循环对称复高斯分布

\end{denotation}
